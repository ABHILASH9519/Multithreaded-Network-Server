cmake_minimum_required(VERSION 3.20)
project(multithreaded_network_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(server_core src/server.cpp)
target_include_directories(server_core PUBLIC include)
target_link_libraries(server_core PUBLIC Threads::Threads)
target_compile_options(server_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(network_server src/main.cpp)
target_link_libraries(network_server PRIVATE server_core)
enable_testing()
add_executable(server_tests tests/server_tests.cpp)
target_link_libraries(server_tests PRIVATE server_core)
add_test(NAME server_tests COMMAND server_tests)
